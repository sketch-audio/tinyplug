set(PLUGIN_TARGET StateDemo_lib) # Generated.

add_library(${PLUGIN_TARGET} STATIC)

# Your plug-in information goes here.
# ---
add_property(${PLUGIN_TARGET} TINY_COMPANY_NAME "tinyplug")
add_property(${PLUGIN_TARGET} TINY_COMPANY_WEBSITE "https://github.com/sketch-audio/tinyplug")
add_property(${PLUGIN_TARGET} TINY_COMPANY_EMAIL "info@tinyplug.com")

# These need to be four-character codes.
add_property(${PLUGIN_TARGET} TINY_FRAMEWORK_CODE "tiny")
add_property(${PLUGIN_TARGET} TINY_MANUFACTURER_CODE "Tiny") # At least one capital letter.
add_property(${PLUGIN_TARGET} TINY_PLUGIN_CODE "stat") # Must be unique per manufacturer code!

add_property(${PLUGIN_TARGET} TINY_PRODUCT_NAME "State Demo")
add_property(${PLUGIN_TARGET} TINY_PLUGIN_NAME "State Demo")
add_property(${PLUGIN_TARGET} TINY_PLUGIN_SHORT_NAME "StateDemo") # Must be <= 16 chars, for Pro Tools.

add_property(${PLUGIN_TARGET} TINY_BASE_FILENAME "StateDemo")
add_property(${PLUGIN_TARGET} TINY_BASE_IDENTIFIER "com.tinyplug.StateDemo")
add_property(${PLUGIN_TARGET} TINY_VERSION_STRING "1.0.0")
add_property(${PLUGIN_TARGET} TINY_BUILD_NUMBER "1")

add_property(${PLUGIN_TARGET} TINY_PLUGIN_WANTS_SIDECHAIN "false")
add_property(${PLUGIN_TARGET} TINY_PLUGIN_CAN_PROCESS_MONO "true")

# AAX
# See: AAX_Enums.h for possible values.
add_property(${PLUGIN_TARGET} TINY_AAX_CATEGORIES "AAX_ePlugInCategory_None")

# CLAP
# See: https://github.com/free-audio/clap/blob/main/include/clap/plugin-features.h
add_property(${PLUGIN_TARGET} TINY_CLAP_DESCRIPTION "Built with tinyplug.")
add_property(${PLUGIN_TARGET} TINY_CLAP_FEATURES "audio-effect;utility") # Don't put mono or stereo, the framework will do that for you!

# AUv2
add_property(${PLUGIN_TARGET} TINY_AUV2_VIEW_CLASS "StateDemo_viewFactory") # Audio Unit must have a unique view factory!
add_property(${PLUGIN_TARGET} TINY_AUV2_TYPE "aufx")

# VST3
# See: https://steinbergmedia.github.io/vst3_doc/vstinterfaces/namespaceSteinberg_1_1Vst_1_1PlugType.html
add_property(${PLUGIN_TARGET} TINY_VST3_SUBCATEGORIES "Fx")

# Presets
# The JSON in presets/ is the source: AUv2, AUv3 and CLAP list it, the editor's buttons load it, and the
# exporters below write it out as presets/out/*.tfx (AAX, copied into the bundle) and *.vstpreset (placed
# by an installer). Rerun them after changing a preset: build and run StateDemo_tfx_exporter and
# StateDemo_vstpreset_exporter.
add_property(${PLUGIN_TARGET} TINY_NATIVE_PRESETS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/presets")
add_property(${PLUGIN_TARGET} TINY_FORMAT_PRESETS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/presets/out")
add_property(${PLUGIN_TARGET} TINY_FIRST_PRESET_NAME "Pulse")
# ---

# Your plug-in's sources
# ---
target_sources(${PLUGIN_TARGET} PRIVATE
    source/models/meters.hpp
    source/models/params.hpp
    source/models/state.hpp
    source/models/view.hpp
    source/editor.cpp
    source/editor.hpp
    source/processor.cpp
    source/processor.hpp
)
# ---

# Generate your plug-in's `plug_info.hpp`
# ---
configure_plug_info(
    ${PLUGIN_TARGET}
    ${CMAKE_CURRENT_BINARY_DIR}/plug_info.hpp
)
configure_models(${PLUGIN_TARGET}) # <tiny_models.hpp> from source/models/
configure_plugin(${PLUGIN_TARGET}) # <tiny_plugin.hpp> from source/
# ---

# The editor's preset buttons, in this order, compiled in from the same JSON the hosts load.
set(_factory_presets Pulse Offbeat Stutter)
set(_entries "")
foreach(_name ${_factory_presets})
    set(_file ${CMAKE_CURRENT_SOURCE_DIR}/presets/${_name}.json)
    file(READ ${_file} _json)
    string(APPEND _entries "    Factory_preset{\"${_name}\", R\"json(${_json})json\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_file})
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/factory_presets.hpp @ONLY CONTENT [=[
// Generated by examples/state_demo/CMakeLists.txt from presets/*.json.
#pragma once

#include <array>
#include <string_view>

namespace tiny::edit {

struct Factory_preset {
    std::string_view name{};
    std::string_view json{};
};

inline constexpr auto factory_presets = std::to_array({
@_entries@});

} // namespace tiny::edit
]=])

target_include_directories(${PLUGIN_TARGET} PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/source
    ${CMAKE_CURRENT_BINARY_DIR} # for plug_info.hpp
)
target_link_libraries(${PLUGIN_TARGET} PUBLIC ${TINY_PLUG_LIB}) # plugin headers expose framework types
target_link_libraries(${PLUGIN_TARGET} PRIVATE ${TINY_PLATFORM_LIB}) # editor may use platform services (dialogs/paths)
target_link_libraries(${PLUGIN_TARGET} PRIVATE tiny::skia) # editor draws with Skia
target_link_libraries(${PLUGIN_TARGET} PRIVATE ${TINY_UI_LIB}) # editor.cpp labels with its pixel font

# Build the wrappers
# ---
make_aax_plugin(${PLUGIN_TARGET})
make_auv2_plugin(${PLUGIN_TARGET}) # Skips build if not Apple.
make_auv3_plugin(${PLUGIN_TARGET})
make_clap_plugin(${PLUGIN_TARGET})
make_vst3_plugin(${PLUGIN_TARGET})
make_web_plugin(${PLUGIN_TARGET})

make_tfx_exporter(${PLUGIN_TARGET} ${CMAKE_CURRENT_SOURCE_DIR}/presets)
make_vstpreset_exporter(${PLUGIN_TARGET} ${CMAKE_CURRENT_SOURCE_DIR}/presets)
# ---
